# 第一部分完整编程示例

这份脚本把前面章节组合成一个可运行的最小游戏：三个自由窗口、二维地图、移动、开门、拾取、日志、对话框、动态布局和终端分辨率切换。

它不是必须逐行背诵的“标准架构”，而是工具接口的完整参考。先运行，再按功能拆开阅读。

## 接线

1. 在 Unity 项目创建 `Assets/Game/Runtime/FirstAsciiRoomModule.cs`。
2. 复制下面代码，文件名与类名必须一致。
3. 把脚本添加到 `TerminalModuleHost` 所在 GameObject。
4. 确认同对象存在 `TerminalDisplay` 和 `TerminalModuleHost`。
5. 进入 Play Mode。

操作：WASD/方向键移动，E 开门，Space 对话，Tab 开关右侧面板，R 切换终端分辨率。

## 完整代码

```csharp
using System.Collections.Generic;
using AsciiTerminal;
using UnityEngine;

namespace YourGame
{
    public sealed class FirstAsciiRoomModule : TerminalModuleBehaviour
    {
        private static readonly TerminalStyle Background = new TerminalStyle(
            new Color32(35, 62, 64, 255),
            new Color32(0, 0, 0, 255));

        private static readonly TerminalStyle Wall = new TerminalStyle(
            new Color32(130, 155, 150, 255),
            new Color32(0, 0, 0, 255));

        private static readonly TerminalStyle Accent = new TerminalStyle(
            new Color32(255, 205, 90, 255),
            new Color32(0, 0, 0, 255));

        private static readonly TerminalStyle Player = new TerminalStyle(
            new Color32(235, 255, 225, 255),
            new Color32(0, 0, 0, 255));

        private readonly List<string> _log = new List<string>();

        private TerminalModuleContext _terminal;
        private TerminalWindow _header;
        private TerminalWindow _mapWindow;
        private TerminalWindow _panel;
        private TerminalWindow _dialog;

        private char[,] _map;
        private Vector2Int _player = new Vector2Int(2, 2);
        private int _items;
        private int _steps;
        private bool _showPanel = true;
        private bool _dialogOpen;
        private string _dialogText = string.Empty;
        private bool _dirty = true;

        public override string ModuleId => "game.first-room";

        public override void OnTerminalOpen(TerminalModuleContext context)
        {
            _terminal = context;

            _header = context.OpenWindow(
                "header",
                new Rect(0f, 0f, 1f, 0.08f),
                new Vector2Int(100, 4),
                layer: 20);

            _mapWindow = context.OpenWindow(
                "map",
                new Rect(0f, 0.08f, 0.72f, 0.92f),
                new Vector2Int(72, 32),
                layer: 0);

            _panel = context.OpenWindow(
                "panel",
                new Rect(0.72f, 0.08f, 0.28f, 0.92f),
                new Vector2Int(32, 32),
                layer: 10);

            _dialog = context.OpenWindow(
                "dialog",
                new Rect(0.12f, 0.66f, 0.76f, 0.25f),
                new Vector2Int(76, 12),
                layer: 50);
            _dialog.Visible = false;

            BuildMap();
            AddLog("LINK ESTABLISHED");
            AddLog("Find the signal and open the gate.");
            ApplyLayout();
            _dirty = true;
        }

        public override void OnTerminalFrame(TerminalFrameContext frame)
        {
            if (!_dirty)
                return;

            DrawHeader();
            DrawMap();
            DrawPanel();
            DrawDialog();
            _dirty = false;
        }

        public override bool OnTerminalInput(TerminalInputContext input)
        {
            if (_dialogOpen)
            {
                if ((input.Kind == TerminalInputKind.Character &&
                     input.Character == ' ') ||
                    (input.Kind == TerminalInputKind.KeyDown &&
                     input.Key == KeyCode.Escape))
                {
                    _dialogOpen = false;
                    _dialog.Visible = false;
                    _dirty = true;
                }

                return true;
            }

            if (input.Kind == TerminalInputKind.KeyDown)
            {
                switch (input.Key)
                {
                    case KeyCode.UpArrow: TryMove(0, -1); return true;
                    case KeyCode.DownArrow: TryMove(0, 1); return true;
                    case KeyCode.LeftArrow: TryMove(-1, 0); return true;
                    case KeyCode.RightArrow: TryMove(1, 0); return true;
                    case KeyCode.Tab: TogglePanel(); return true;
                }
            }

            if (input.Kind != TerminalInputKind.Character)
                return false;

            switch (char.ToLowerInvariant(input.Character))
            {
                case 'w': TryMove(0, -1); return true;
                case 's': TryMove(0, 1); return true;
                case 'a': TryMove(-1, 0); return true;
                case 'd': TryMove(1, 0); return true;
                case 'e': Interact(); return true;
                case ' ': OpenDialog("A weak carrier wave answers."); return true;
                case 'r': ToggleResolution(); return true;
                default: return false;
            }
        }

        public override void OnTerminalClose()
        {
            CloseWindow("dialog", ref _dialog);
            CloseWindow("panel", ref _panel);
            CloseWindow("map", ref _mapWindow);
            CloseWindow("header", ref _header);
        }

        private void BuildMap()
        {
            string[] rows =
            {
                "########################",
                "#..........#...........#",
                "#..S.......+.......E...#",
                "#..........#...........#",
                "########################"
            };

            _map = new char[rows[0].Length, rows.Length];
            for (int y = 0; y < rows.Length; y++)
            for (int x = 0; x < rows[y].Length; x++)
                _map[x, y] = rows[y][x];
        }

        private void TryMove(int dx, int dy)
        {
            Vector2Int next = _player + new Vector2Int(dx, dy);
            if (!InsideMap(next))
                return;

            char tile = _map[next.x, next.y];
            if (tile == '#' || tile == '+')
            {
                AddLog(tile == '+' ? "The gate is closed." : "A wall blocks the signal.");
                _dirty = true;
                return;
            }

            _player = next;
            _steps++;

            if (tile == 'S')
            {
                _items++;
                _map[next.x, next.y] = '.';
                AddLog("Signal fragment recovered.");
            }
            else if (tile == 'E')
            {
                OpenDialog(_items > 0
                    ? "The exit recognizes your signal."
                    : "The exit remains silent.");
            }

            _dirty = true;
        }

        private void Interact()
        {
            Vector2Int[] directions =
            {
                Vector2Int.up,
                Vector2Int.down,
                Vector2Int.left,
                Vector2Int.right
            };

            for (int i = 0; i < directions.Length; i++)
            {
                Vector2Int p = _player + directions[i];
                if (InsideMap(p) && _map[p.x, p.y] == '+')
                {
                    _map[p.x, p.y] = '/';
                    AddLog("The gate opens.");
                    _dirty = true;
                    return;
                }
            }

            AddLog("Nothing nearby responds.");
            _dirty = true;
        }

        private void TogglePanel()
        {
            _showPanel = !_showPanel;
            ApplyLayout();
            _dirty = true;
        }

        private void ApplyLayout()
        {
            float mapWidth = _showPanel ? 0.72f : 1f;
            _terminal.SetWindowLayout(
                _mapWindow,
                new Rect(0f, 0.08f, mapWidth, 0.92f));

            _panel.Visible = _showPanel;
            if (_showPanel)
            {
                _terminal.SetWindowLayout(
                    _panel,
                    new Rect(0.72f, 0.08f, 0.28f, 0.92f));
            }
        }

        private void ToggleResolution()
        {
            Vector2Int current = _terminal.Terminal.Resolution;
            bool useLarge = current.x < 300;
            _terminal.Terminal.SetResolution(
                useLarge ? 400 : 160,
                useLarge ? 200 : 90,
                TerminalResizeMode.ScaleWindowBounds);
            ApplyLayout();
            AddLog(useLarge ? "Resolution 400 x 200" : "Resolution 160 x 90");
            _dirty = true;
        }

        private void OpenDialog(string text)
        {
            _dialogText = text;
            _dialogOpen = true;
            _dialog.Visible = true;
            _dirty = true;
        }

        private void DrawHeader()
        {
            TerminalCanvas c = _header.Canvas;
            c.Clear(Background);
            c.DrawBox(new RectInt(0, 0, c.Width, c.Height), Wall);
            c.Write(2, 1, "ASH TERMINAL / ROOM 01", Accent);
        }

        private void DrawMap()
        {
            TerminalCanvas c = _mapWindow.Canvas;
            c.Clear(Background);
            c.DrawBox(new RectInt(0, 0, c.Width, c.Height), Wall);

            Vector2Int origin = new Vector2Int(3, 3);
            for (int y = 0; y < _map.GetLength(1); y++)
            for (int x = 0; x < _map.GetLength(0); x++)
            {
                char tile = _map[x, y];
                TerminalStyle style = tile == '#' ? Wall : Background;
                if (tile == 'S' || tile == 'E' || tile == '+')
                    style = Accent;
                c.Put(origin.x + x, origin.y + y, tile, style);
            }

            c.Put(origin.x + _player.x, origin.y + _player.y, '@', Player);
        }

        private void DrawPanel()
        {
            if (!_showPanel)
                return;

            TerminalCanvas c = _panel.Canvas;
            c.Clear(Background);
            c.DrawBox(new RectInt(0, 0, c.Width, c.Height), Wall);
            c.Write(2, 2, $"SIGNALS {_items}", Accent);
            c.Write(2, 3, $"STEPS   {_steps}", Wall);
            c.Write(2, 5, "LOG", Accent);

            int first = Mathf.Max(0, _log.Count - 6);
            for (int i = first; i < _log.Count; i++)
            {
                c.Write(2, 7 + i - first, _log[i], Wall,
                    maxWidth: c.Width - 4, wrap: true);
            }
        }

        private void DrawDialog()
        {
            if (!_dialogOpen)
                return;

            TerminalCanvas c = _dialog.Canvas;
            c.Clear(Background);
            c.DrawBox(new RectInt(0, 0, c.Width, c.Height), Accent);
            c.Write(3, 2, "CARRIER", Accent);
            c.Write(3, 4, _dialogText, Player, c.Width - 6, true);
            c.Write(3, c.Height - 2, "[SPACE] CLOSE", Wall);
        }

        private void AddLog(string message)
        {
            _log.Add(message);
            if (_log.Count > 20)
                _log.RemoveAt(0);
        }

        private bool InsideMap(Vector2Int p)
        {
            return p.x >= 0 && p.y >= 0 &&
                   p.x < _map.GetLength(0) &&
                   p.y < _map.GetLength(1);
        }

        private void CloseWindow(string id, ref TerminalWindow window)
        {
            if (window == null)
                return;
            _terminal.CloseWindow(id);
            window = null;
        }
    }
}
```

## 代码按职责阅读

- `OnTerminalOpen`：只负责申请窗口和初始化。
- `OnTerminalInput`：只改变事实状态。
- `Draw...`：把状态变成字符，不负责游戏规则。
- `ApplyLayout`：集中管理窗口位置与占比。
- `OnTerminalClose`：释放本模块创建的窗口。

## 建议做的三个修改

1. 把门改成需要拿到信号后才能打开。
2. 把右侧面板改为浮动在地图上方的透明窗口。
3. 增加第二个房间，但不要把新房间逻辑写进绘制函数。

完成这三个修改后，第一部分的主要接口就已经实际用过了。
